-- ---------------------------------------------------------------
-- TABLES
-- ---------------------------------------------------------------
-- Teachers
CREATE TABLE teachers (
    teacher_id INT PRIMARY KEY,
    name       VARCHAR(100) NOT NULL
);

-- What each teacher can teach (a range of classes per subject)
CREATE TABLE teacher_classes (
    id         INT AUTO_INCREMENT PRIMARY KEY,
    teacher_id INT          NOT NULL,
    subject    VARCHAR(60)  NOT NULL,
    class_from TINYINT      NOT NULL,
    class_to   TINYINT      NOT NULL,
    FOREIGN KEY (teacher_id) REFERENCES teachers(teacher_id),
    CHECK (class_from BETWEEN 1 AND 12 AND class_to BETWEEN 1 AND 12 AND class_from <= class_to)
);

-- The actual class groups (one row per section)
CREATE TABLE classes (
    class_id    INT AUTO_INCREMENT PRIMARY KEY,
    class_level TINYINT     NOT NULL,
    section     CHAR(1)     NOT NULL,
    stream      VARCHAR(20) NOT NULL DEFAULT 'ALL',   -- ALL for classes 1-10; PCM / PCB / COMMERCE for 11-12
    UNIQUE (class_level, section),
    CHECK (class_level BETWEEN 1 AND 12)
);

-- Weekly periods each subject needs, per grade and stream (same for every section)
CREATE TABLE subject_periods (
    class_level      TINYINT     NOT NULL,
    stream           VARCHAR(20) NOT NULL DEFAULT 'ALL',
    subject          VARCHAR(60) NOT NULL,
    periods_per_week TINYINT     NOT NULL,
    PRIMARY KEY (class_level, stream, subject),
    CHECK (periods_per_week > 0)
);

-- The 30 weekly slots (Mon-Fri, periods 1-6, 60 minutes each)
CREATE TABLE time_slots (
    slot_id          INT AUTO_INCREMENT PRIMARY KEY,
    day_no           TINYINT     NOT NULL,
    day_name         CHAR(3)     NOT NULL,
    period_no        TINYINT     NOT NULL,
    duration_minutes TINYINT     NOT NULL DEFAULT 60,
    UNIQUE (day_no, period_no)
);

-- The generated timetable (what the solver writes into).
-- The two unique keys make clashes impossible:
--   PRIMARY KEY (class_id, slot_id)  -> a class has only one subject in a slot
--   UNIQUE (teacher_id, slot_id)     -> a teacher is in only one class in a slot
CREATE TABLE timetable (
    class_id   INT         NOT NULL,
    slot_id    INT         NOT NULL,
    subject    VARCHAR(60) NOT NULL,
    teacher_id INT         NOT NULL,
    PRIMARY KEY (class_id, slot_id),
    UNIQUE KEY uq_teacher_slot (teacher_id, slot_id),
    FOREIGN KEY (class_id)   REFERENCES classes(class_id),
    FOREIGN KEY (slot_id)    REFERENCES time_slots(slot_id),
    FOREIGN KEY (teacher_id) REFERENCES teachers(teacher_id)
);

INSERT INTO time_slots (day_no, day_name, period_no)
SELECT d.day_no, d.day_name, p.period_no
FROM (SELECT 1 AS day_no, 'Mon' AS day_name UNION ALL SELECT 2, 'Tue' UNION ALL SELECT 3, 'Wed'
      UNION ALL SELECT 4, 'Thu' UNION ALL SELECT 5, 'Fri') d
CROSS JOIN (SELECT 1 AS period_no UNION ALL SELECT 2 UNION ALL SELECT 3
            UNION ALL SELECT 4 UNION ALL SELECT 5 UNION ALL SELECT 6) p
ORDER BY d.day_no, p.period_no;

from subject_periods, teacher_classes, teachers,classes
where subject_periods.subject = teacher_classes.subject
and subject_periods.class_level between teacher_classes.class_from and teacher_classes.class_to
and teacher_classes.teacher_id = teachers.teacher_id
and classes.class_level = subject_periods.class_level;

select sum(periods_per_week),class_level,stream
from subject_periods
group by class_level,stream;
